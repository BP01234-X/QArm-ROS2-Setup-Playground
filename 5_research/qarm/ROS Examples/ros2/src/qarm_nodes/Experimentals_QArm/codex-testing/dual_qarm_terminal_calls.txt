#!/usr/bin/env bash

# Dual QArm terminal call reference
#
# What this file is:
# - A commented set of terminal commands you can copy and run manually.
# - It shows how to run two QArms independently with the current ROS 2 stack.
# - It also shows how to send goals to both arms at the same time from the terminal.
#
# Important limitation right now:
# - The experimental differential-kinematics server in Experimentals_QArm/move_qarm_server.py
#   cannot run end-to-end yet because MoveQArm.action does not currently define a "mode" field.
# - So the commands below use the current installed task-mode nodes from qarm_nodes.
#
# Core idea:
# - device_id selects which physical QArm the hardware node opens through Quanser HIL.
# - ROS topic remaps keep the two arms from talking over the same topics.
# - A namespace keeps each move_qarm action separate:
#     /qarm0/move_qarm
#     /qarm1/move_qarm
#
# Recommended workflow:
# 1. Open one terminal per long-running node.
# 2. Source the QArm environment in each terminal.
# 3. Copy the command block you need.
#
# ---------------------------------------------------------------------------
# 0. Source the environment in every terminal you use
# ---------------------------------------------------------------------------
#
  cd ~/Documents/QArm-ROS2-Setup-Playground
  source ~/qarm_env/scripts/enter_qarm_playground.sh
  source install/setup.bash
#
# ---------------------------------------------------------------------------
# 1. Start hardware node for physical QArm with device_id=0
# ---------------------------------------------------------------------------
#
# What this does:
# - Opens the physical QArm identified by Quanser as device 0.
# - Publishes that arm's joint states on /qarm0/joint_states.
# - Subscribes for commands on /qarm0/joint_position_cmd and /qarm0/led_cmd.
#
 ros2 run qarm_nodes qarm_hardware --ros-args \
   -r __node:=qarm0_hardware \
   -p device_id:=0 \
   -r /qarm/joint_states:=/qarm0/joint_states \
   -r /qarm/joint_position_cmd:=/qarm0/joint_position_cmd \
   -r /qarm/led_cmd:=/qarm0/led_cmd \
   -r qarm/diagnostics:=/qarm0/diagnostics
#
# ---------------------------------------------------------------------------
# 2. Start hardware node for physical QArm with device_id=1
# ---------------------------------------------------------------------------
#
# What this does:
# - Opens the physical QArm identified by Quanser as device 1.
# - Publishes that arm's joint states on /qarm1/joint_states.
# - Subscribes for commands on /qarm1/joint_position_cmd and /qarm1/led_cmd.
#
 ros2 run qarm_nodes qarm_hardware --ros-args \
   -r __node:=qarm1_hardware \
   -p device_id:=1 \
   -r /qarm/joint_states:=/qarm1/joint_states \
   -r /qarm/joint_position_cmd:=/qarm1/joint_position_cmd \
   -r /qarm/led_cmd:=/qarm1/led_cmd \
   -r qarm/diagnostics:=/qarm1/diagnostics
#
# ---------------------------------------------------------------------------
# 3. Start move_qarm action server for QArm 0
# ---------------------------------------------------------------------------
#
# What this does:
# - Creates the action server /qarm0/move_qarm.
# - Reads joint states from /qarm0/joint_states.
# - Sends position and LED commands only to QArm 0 topics.
#
 ros2 run qarm_nodes move_qarm_server --ros-args \
   -r __node:=qarm0_move_server \
   -r __ns:=/qarm0 \
   -r qarm/joint_states:=/qarm0/joint_states \
   -r qarm/joint_position_cmd:=/qarm0/joint_position_cmd \
   -r qarm/led_cmd:=/qarm0/led_cmd

# ---------------------------------------------------------------------------
# 4. Start move_qarm action server for QArm 1
# ---------------------------------------------------------------------------
#
# What this does:
# - Creates the action server /qarm1/move_qarm.
# - Reads joint states from /qarm1/joint_states.
# - Sends position and LED commands only to QArm 1 topics.
#
 ros2 run qarm_nodes move_qarm_server --ros-args \
   -r __node:=qarm1_move_server \
   -r __ns:=/qarm1 \
   -r qarm/joint_states:=/qarm1/joint_states \
   -r qarm/joint_position_cmd:=/qarm1/joint_position_cmd \
   -r qarm/led_cmd:=/qarm1/led_cmd
#
# ---------------------------------------------------------------------------
# 5. Move only QArm 0 from the terminal
# ---------------------------------------------------------------------------
#
# What this does:
# - Runs one action client under namespace /qarm0.
# - Sends a task-space goal only to /qarm0/move_qarm.
#
 ros2 run qarm_nodes move_qarm_client --ros-args \
   -r __node:=qarm0_move_client \
   -r __ns:=/qarm0 \
   -p goal_pose:="[0.30, 0.10, 0.25, 0.00]"
#
# ---------------------------------------------------------------------------
# 6. Move only QArm 1 from the terminal
# ---------------------------------------------------------------------------
#
# What this does:
# - Runs one action client under namespace /qarm1.
# - Sends a task-space goal only to /qarm1/move_qarm.
#
 ros2 run qarm_nodes move_qarm_client --ros-args \
   -r __node:=qarm1_move_client \
   -r __ns:=/qarm1 \
   -p goal_pose:="[0.30, -0.10, 0.25, 0.00]"
#
# ---------------------------------------------------------------------------
# 7. Move both QArms at the same time from one terminal
# ---------------------------------------------------------------------------
#
# What this does:
# - Starts one client for QArm 0 in the background.
# - Starts one client for QArm 1 in the background.
# - "wait" keeps the shell open until both client processes finish.
#
 ros2 run qarm_nodes move_qarm_client --ros-args \
   -r __node:=qarm0_move_client \
   -r __ns:=/qarm0 \
   -p goal_pose:="[0.30, 0.10, 0.25, 0.00]" &

 ros2 run qarm_nodes move_qarm_client --ros-args \
   -r __node:=qarm1_move_client \
   -r __ns:=/qarm1 \
   -p goal_pose:="[0.30, -0.10, 0.25, 0.00]" &


######################

 ros2 run qarm_nodes move_qarm_client --ros-args \
   -r __node:=qarm0_move_client \
   -r __ns:=/qarm0 \
   -p goal_pose:="[0.30, 0.10, 0.45, 0.00]" &

 ros2 run qarm_nodes move_qarm_client --ros-args \
   -r __node:=qarm1_move_client \
   -r __ns:=/qarm1 \
   -p goal_pose:="[0.30, -0.10, 0.45, 0.00]" &




# wait
#
# ---------------------------------------------------------------------------
# 8. Optional quick LED-only check for each arm
# ---------------------------------------------------------------------------
#
# What this does:
# - Publishes directly to each arm's LED topic without using the action server.
# - Useful to confirm your remaps before commanding motion.
#
# ros2 topic pub --once /qarm0/led_cmd std_msgs/msg/Float64MultiArray "{data: [0.0, 1.0, 0.0]}"
# ros2 topic pub --once /qarm1/led_cmd std_msgs/msg/Float64MultiArray "{data: [1.0, 0.0, 1.0]}"
#
# ---------------------------------------------------------------------------
# 9. Future experimental differential-mode note
# ---------------------------------------------------------------------------
#
# The experimental server/client under Experimentals_QArm are not ready to call yet.
# Before you can use them, you need to:
# - add a "mode" field and constants to qarm_interfaces/action/MoveQArm.action
# - rebuild qarm_interfaces and qarm_nodes
# - decide whether to install the experimental files as console entry points
#
# Once that is done, you can reuse the same remap pattern shown above.
